  function $(s, r) { return (r || document).querySelector(s); }
  function $$(s, r) { return Array.prototype.slice.call((r || document).querySelectorAll(s)); }

  /* Split the name into characters */
  $$('[data-split]').forEach(function (el) {
    var t = el.textContent;
    el.textContent = '';
    for (var i = 0; i < t.length; i++) {
      var s = document.createElement('span');
      s.className = 'ch';
      s.setAttribute('aria-hidden', 'true');
      s.textContent = t.charAt(i);
      el.appendChild(s);
    }
  });

  /* Typed line */
  var lines = ['Building across the stack', 'C++ · Java · Kotlin · Flutter', 'Next.js · Prisma · Supabase', 'Every bug gets a happy ending'];
  var tEl = $('#typed');
  if (reduce) {
    tEl.textContent = lines[0];
  } else {
    var li = 0, ci = 0, del = false;
    (function tick() {
      var cur = lines[li];
      ci += del ? -1 : 1;
      tEl.textContent = cur.slice(0, ci);
      var d = del ? 24 : 55;
      if (!del && ci === cur.length) { del = true; d = 1500; }
      else if (del && ci === 0) { del = false; li = (li + 1) % lines.length; d = 380; }
      setTimeout(tick, d);
    })();
  }

  /* HUD + scroll state */
  var secs = $$('[data-scene]');
  var sceneEl = $('#scene'), depthEl = $('#depth'), barEl = $('#bar');
  function onScroll() {
    var h = document.documentElement.scrollHeight - window.innerHeight;
    var p = h > 0 ? Math.min(1, Math.max(0, window.scrollY / h)) : 0;
    state.p = p;
    barEl.style.transform = 'scaleX(' + p + ')';
    depthEl.textContent = String(Math.round(p * 660)).padStart(3, '0');
    var cur = secs[0];
    for (var i = 0; i < secs.length; i++) {
      if (secs[i].getBoundingClientRect().top < window.innerHeight * 0.5) cur = secs[i];
    }
    sceneEl.textContent = cur.getAttribute('data-scene');
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  window.addEventListener('resize', onScroll);
  onScroll();

  /* GSAP choreography */
  if (window.gsap && window.ScrollTrigger && !reduce) {
    gsap.registerPlugin(ScrollTrigger);
    gsap.from('.ch', { yPercent: 110, rotateX: -70, opacity: 0.25, duration: 1.1, ease: 'power4.out', stagger: 0.045, delay: 0.1 });
    gsap.from('.hero-fade', { y: 22, opacity: 0.2, duration: 1, ease: 'power3.out', stagger: 0.12, delay: 0.6 });
    $$('[data-reveal]').forEach(function (el) {
      gsap.from(el, { y: 56, opacity: 0.2, duration: 0.9, ease: 'power3.out', scrollTrigger: { trigger: el, start: 'top 90%', toggleActions: 'play none none none' } });
    });
    $$('[data-drift]').forEach(function (el) {
      var d = parseFloat(el.getAttribute('data-drift'));
      gsap.fromTo(el, { xPercent: -d }, { xPercent: d, ease: 'none', scrollTrigger: { trigger: el.parentNode, start: 'top bottom', end: 'bottom top', scrub: true } });
    });
  }

  /* 3D tilt on stack layers */
  $$('.tilt').forEach(function (el) {
    el.addEventListener('pointermove', function (e) {
      if (e.pointerType === 'touch' || reduce) return;
      var r = el.getBoundingClientRect();
      var x = (e.clientX - r.left) / r.width - 0.5;
      var y = (e.clientY - r.top) / r.height - 0.5;
      el.style.transform = 'perspective(900px) rotateX(' + (-y * 12).toFixed(2) + 'deg) rotateY(' + (x * 14).toFixed(2) + 'deg) translateZ(8px)';
    });
    el.addEventListener('pointerleave', function () { el.style.transform = ''; });
  });

  /* Copy email */
  var cb = $('#copy'), em = $('#email');
  function selectEmail() {
    var r = document.createRange(); r.selectNodeContents(em);
    var s = window.getSelection(); s.removeAllRanges(); s.addRange(r);
  }
  cb.addEventListener('click', function () {
    var txt = em.textContent.trim();
    function done() { cb.textContent = 'Copied'; setTimeout(function () { cb.textContent = 'Copy'; }, 1600); }
    function fallback() { selectEmail(); cb.textContent = 'Selected'; setTimeout(function () { cb.textContent = 'Copy'; }, 1600); }
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(txt).then(done, fallback);
    } else { fallback(); }
  });

  /* WebGL flight */
  var cv = $('#gl');
  var renderer = null;
  try {
    if (!window.THREE) throw new Error('no three');
    renderer = new THREE.WebGLRenderer({ canvas: cv, antialias: true, alpha: true });
  } catch (err) {
    document.body.classList.add('no-gl');
    return;
  }
  renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, 2));
  var scene = new THREE.Scene();
  scene.fog = new THREE.FogExp2(0x03050a, 0.02);
  var cam = new THREE.PerspectiveCamera(60, 1, 0.1, 220);
  var SIGNAL = 0x46f0d2, EMBER = 0xff6b3d, ICE = 0xe6edf7;

  /* Core */
  var core = new THREE.Group();
  core.add(new THREE.Mesh(new THREE.IcosahedronGeometry(1.7, 1), new THREE.MeshBasicMaterial({ color: 0x061018 })));
  var wire = new THREE.LineSegments(new THREE.WireframeGeometry(new THREE.IcosahedronGeometry(1.72, 1)), new THREE.LineBasicMaterial({ color: SIGNAL, transparent: true, opacity: 0.9 }));
  var shell = new THREE.LineSegments(new THREE.WireframeGeometry(new THREE.IcosahedronGeometry(2.7, 2)), new THREE.LineBasicMaterial({ color: SIGNAL, transparent: true, opacity: 0.16 }));
  var inner = new THREE.LineSegments(new THREE.WireframeGeometry(new THREE.OctahedronGeometry(0.9, 0)), new THREE.LineBasicMaterial({ color: EMBER, transparent: true, opacity: 0.95 }));
  core.add(wire, shell, inner);
  scene.add(core);

  /* Particles along the flight path */
  function field(n, size, color, op) {
    var pos = new Float32Array(n * 3);
    for (var i = 0; i < n; i++) {
      var a = Math.random() * Math.PI * 2, r = 3 + Math.random() * 16;
      pos[i * 3] = Math.cos(a) * r;
      pos[i * 3 + 1] = Math.sin(a) * r * 0.7;
      pos[i * 3 + 2] = 12 - Math.random() * 100;
    }
    var g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.BufferAttribute(pos, 3));
    var pts = new THREE.Points(g, new THREE.PointsMaterial({ color: color, size: size, sizeAttenuation: true, transparent: true, opacity: op, depthWrite: false }));
    scene.add(pts);
    return pts;
  }
  var dust = field(2600, 0.06, ICE, 0.8);
  var sparks = field(260, 0.12, EMBER, 0.9);

  /* Rings the camera flies through */
  var rings = [];
  for (var i = 0; i < 10; i++) {
    var m = new THREE.Mesh(
      new THREE.TorusGeometry(3 + (i % 3) * 0.7, 0.014, 8, 140),
      new THREE.MeshBasicMaterial({ color: i % 4 === 3 ? EMBER : SIGNAL, transparent: true, opacity: 0.55 })
    );
    m.position.z = -8 - i * 6.2;
    m.rotation.set(Math.random() * 0.8, Math.random() * 0.8, Math.random() * Math.PI);
    m.userData.s = (Math.random() * 0.4 + 0.15) * (i % 2 ? 1 : -1);
    scene.add(m); rings.push(m);
  }

  /* Destination knot */
  var knot = new THREE.Mesh(new THREE.TorusKnotGeometry(1.3, 0.34, 180, 14), new THREE.MeshBasicMaterial({ color: EMBER, wireframe: true, transparent: true, opacity: 0.85 }));
  knot.position.set(0, 0, -66);
  scene.add(knot);

  function resize() {
    var w = window.innerWidth, h = window.innerHeight;
    renderer.setSize(w, h, false);
    cam.aspect = w / h; cam.updateProjectionMatrix();
    core.position.set(w > 860 ? 2.6 : 0, w > 860 ? 0 : -1.2, 0);
    core.scale.setScalar(w > 860 ? 1 : 0.8);
  }
  window.addEventListener('resize', resize);
  resize();
  window.addEventListener('pointermove', function (e) {
    state.mx = (e.clientX / window.innerWidth) * 2 - 1;
    state.my = (e.clientY / window.innerHeight) * 2 - 1;
  }, { passive: true });

  var cur = state.p, mx = 0, my = 0;
  var clock = new THREE.Clock();
  var TRAVEL = 70;
  function frame() {
    var t = reduce ? 0 : clock.getElapsedTime();
    cur += (state.p - cur) * (reduce ? 1 : 0.07);
    mx += (state.mx - mx) * 0.05; my += (state.my - my) * 0.05;
    cam.position.z = 9 - cur * TRAVEL;
    cam.position.x = Math.sin(cur * Math.PI * 3) * 1.1 + mx * 0.8;
    cam.position.y = Math.cos(cur * Math.PI * 2) * 0.5 - my * 0.5;
    cam.lookAt(mx * 0.5, -my * 0.3, cam.position.z - 12);
    core.rotation.y = t * 0.26 + cur * 5;
    core.rotation.x = t * 0.11;
    inner.rotation.y = -t * 0.8; inner.rotation.z = t * 0.4;
    shell.rotation.y = -t * 0.08;
    for (var i = 0; i < rings.length; i++) rings[i].rotation.z += rings[i].userData.s * 0.004 * (reduce ? 0 : 1);
    knot.rotation.x = t * 0.3 + cur * 3; knot.rotation.y = t * 0.45;
    dust.rotation.z = t * 0.01;
    renderer.render(scene, cam);
    window.requestAnimationFrame(frame);
  }
  frame();
})();
</script>
